TRY_REPLACE:grandmaster_elective = {
	include_ruler_siblings = yes
	all_in_country = yes
	through_female = no
	allow_female = no
	ignore_ruler = yes
	allow_foreign_ruler = no
	custom_tags = { empty_heir_religion_law }
	allowed_estates = { nobles_estate clergy_estate }
	heir_is_allowed = {}
	potential = {
		has_reform = government_reform:military_order_reform
	}
	locked = {
		has_reform = government_reform:military_order_reform
	}
	calc = {
		if = {
			limit = {
				exists = root
			}
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = character_age
			}
			if = { 
				limit = { modifier:blocked_from_being_ruler = yes }
				add = {
					desc = "HEIR_SELECTION_IS_UNSUITED_TO_RULE_DESC"
					value = -1000 
				}
			}
		}
		if = {
			limit = {
				exists = root
			}
			add = {
				desc = "HEIR_SELECTION_MIL_DESC"
				value = root.mil
			}
			if = { 
				limit = { is_general = yes }
				add = {
					desc = "HEIR_SELECTION_GENERAL_DESC"
					value = 10 
				}
			}
		# Here it goes the description for explaining a heir gets +1 points per year.
		if = {
			limit = {
				not = { exists = root }
			}
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = 1
			}
			add = {
				desc = "HEIR_SELECTION_MIL_DESC"
				value = 1
			}
			add = {
				desc = "HEIR_SELECTION_GENERAL_DESC"
				value = 10
			}
		}
	}
}
